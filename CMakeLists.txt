cmake_minimum_required(VERSION 3.20)
project(TWFC_PlayerMipFix LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with MSVC for Win32: cmake -A Win32 -S . -B build")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(TWFC_PlayerMipFix SHARED src/dll.cpp src/patch.cpp src/exports.def)
target_compile_features(TWFC_PlayerMipFix PRIVATE cxx_std_17)
target_compile_options(TWFC_PlayerMipFix PRIVATE /W4 /WX)
add_executable(patch_tests tests/patch_tests.cpp src/patch.cpp)
target_compile_features(patch_tests PRIVATE cxx_std_17)
target_compile_options(patch_tests PRIVATE /W4 /WX)
enable_testing()
add_test(NAME patch_validation COMMAND patch_tests)
